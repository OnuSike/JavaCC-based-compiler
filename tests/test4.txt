class Main {
  public static void main ( String [] args ) {
    System.out.println ( new Calc ( ).run ( 5 ) );
  }
}

class Calc {
  int base;
  int [] data;
  boolean ok;

  public int run ( int n ) {
    int i;
    int total;
    Calc other;
    i = 0;
    total = 0;
    data = new int [ n + 1 ];
    while ( i < n && !ok ) {
      data [ i ] = i * 2 + base;
      total = total + data [ i ] / 2 - 1;
      i = i + 1;
    }
    other = new Calc ( );
    if ( total >= data.length || ( i == n && this.check ( i ) ) )
      total = other.run ( total - 1 );
    else
      total = ( total + 1 ) * 2;
    return total;
  }

  public boolean check ( int v ) {
    return v != 0 && !( v < 0 );
  }
}
